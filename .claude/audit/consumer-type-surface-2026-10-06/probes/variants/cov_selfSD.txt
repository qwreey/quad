type Epoch = { read Revision: number }
type EpochSet = { [Epoch]: true }
export type Observer = { read __quadObserver: true, read Subscribed: boolean, Subscribe: (self: Observer) -> Observer }
export type ObserverFn<T> = (targetState: StateData<T>, self: Observer, emitFrom: (Epoch | EpochSet)?) -> ()
export type GateEmit = (commit: boolean?) -> boolean
export type GateSetup = (emit: GateEmit) -> () -> ()
export type StateData<T> = { read __quadState: true, read __quadStateValue: T, read Get: (self: StateData<T>) -> T }
export type State<T> = StateData<T> & {
	read Compute: <U>(self: StateData<T>, fn: (self: StateData<T>, previous: U?, ...any) -> U, ...any) -> State<U>,
	read Depend: (self: StateData<T>, ...any) -> State<T>,
	read Apply: (<U>(self: StateData<T>, factory: (State<T>) -> U) -> U) & ((self: StateData<T>, factory: { __apply: (self: StateData<T>, state: any) -> any }) -> any),
	read Observer: (self: StateData<T>, fn: ObserverFn<T>?) -> Observer,
	read Gate: (self: StateData<T>, setup: GateSetup) -> State<T>,
}
export type Source<T> = State<T> & {
	read Revision: number,
	read Set: (self: Source<T>, v: T) -> Source<T>,
	read SetIfChanged: (self: Source<T>, v: T) -> Source<T>,
	read SetAndDispose: (self: Source<T>, v: T) -> Source<T>,
	read Emit: (self: Source<T>) -> Source<T>,
}
