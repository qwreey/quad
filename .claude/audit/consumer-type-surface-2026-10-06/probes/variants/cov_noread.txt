type Epoch = { read Revision: number }
type EpochSet = { [Epoch]: true }
export type Observer = { read __quadObserver: true, read Subscribed: boolean, Subscribe: (self: Observer) -> Observer }
export type ObserverFn<T> = (targetState: StateData<T>, self: Observer, emitFrom: (Epoch | EpochSet)?) -> ()
export type GateEmit = (commit: boolean?) -> boolean
export type GateSetup = (emit: GateEmit) -> () -> ()
export type StateData<T> = { read __quadState: true, read __quadStateValue: T, Get: (self: any) -> T }
export type State<T> = StateData<T> & {
	Compute: <U>(self: any, fn: (self: StateData<T>, previous: U?, ...any) -> U, ...any) -> State<U>,
	Depend: (self: any, ...any) -> State<T>,
	Apply: (<U>(self: any, factory: (State<T>) -> U) -> U) & ((self: any, factory: { __apply: (self: any, state: any) -> any }) -> any),
	Observer: (self: any, fn: ObserverFn<T>?) -> Observer,
	Gate: (self: any, setup: GateSetup) -> State<T>,
}
export type Source<T> = State<T> & {
	Revision: number,
	Set: (self: any, v: T) -> Source<T>,
	SetIfChanged: (self: any, v: T) -> Source<T>,
	SetAndDispose: (self: any, v: T) -> Source<T>,
	Emit: (self: any) -> Source<T>,
}
